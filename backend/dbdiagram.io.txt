Table users {
  id uuid [primary key]

  name varchar
  email varchar [unique]
  email_verified_at timestamp [null]
  password varchar
  rememberToken varchar
  role enum
  user_invite_code varchar [unique]

  created_at timestamp
  updated_at timestamp
}

Table weddings {
  id uuid [primary key]

  created_by varchar [ref: > users.id, not null]

  name varchar
  date timestamp

  created_at timestamp
  updated_at timestamp
  deleted_at timestamp
}

Table wedding_user {
  id uuid

  user_id varchar [ref: > users.id, not null]
  wedding_id varchar [ref: > weddings.id, not null]

  created_at timestamp
  updated_at timestamp
}

Table _tables {
  id uuid [primary key]

  wedding_id varchar [ref: > weddings.id, not null]
  created_by varchar [ref: > users.id, not null]

  name varchar
  capacity int
  position_x float
  position_y float
  width float
  height float
  radius float

  created_at timestamp
  updated_at timestamp
  deleted_at timestamp
}

Table addresses {
  id uuid [primary key]

  street varchar
  house_number varchar
  local_number varchar
  city varchar
  postal_code varchar
}

Table guests {
  id uuid [primary key]

  table_id varchar [ref: > _tables.id, not null]
  created_by varchar [ref: > users.id, not null]
  address_id varchar [ref: > addresses.id, not null]
  main_guest_id varchar [ref: >? guests.id] // kto jest głowną osobą tej osoby tow.

  name varchar
  surname varchar
  acquaintance_type enum //rodzaj znajomosci
  child bool
  groom_guest bool //gość pana młodego
  invitation_prepared bool
  invitation_delivered bool
  confirmed_attendance bool
  diet enum
  table_position_x float
  table_position_y float

  created_at timestamp
  updated_at timestamp
  deleted_at timestamp
}

Table finances {
  id uuid [primary key]

  wedding_id varchar [ref: > weddings.id, not null]
  created_by varchar [ref: > users.id, not null]

  name varchar
  amount double
  category varchar [null]
  contact varchar [null]
  payment_deadline timestamp [null]
  payment_status varchar [null]
  _note text [null]

  created_at timestamp
  updated_at timestamp
  deleted_at timestamp
}

Table calendars {
  id uuid [primary key]

  wedding_id varchar [ref: > weddings.id, not null]
  created_by varchar [ref: > users.id, not null]

  name varchar
  color varchar
  wedding_day_calendar bool

  created_at timestamp
  updated_at timestamp
  deleted_at timestamp
}

Table calendar_event {
  id uuid [primary key]

  calendar_id varchar [ref: > calendars.id, not null]
  created_by varchar [ref: > users.id, not null]

  date timestamp

  created_at timestamp
  updated_at timestamp
  deleted_at timestamp
}
